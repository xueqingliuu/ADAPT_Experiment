# ---------- Helper: NB2 alpha via method-of-moments ----------

def estimate_alpha_moment_nb2(y, X, alpha_init=0.5, min_alpha=1e-6):
    """
    Estimate NB2 dispersion alpha via method of moments, using a provisional GLM NB fit:
    Var(Y|X) = mu + alpha * mu^2.
    """
    fam_init = sm.families.NegativeBinomial(alpha=alpha_init)
    glm_nb_init = sm.GLM(y, X, family=fam_init).fit()

    mu_hat = glm_nb_init.fittedvalues

    resid_sq = (y - mu_hat) ** 2
    num = np.sum(resid_sq - mu_hat)
    den = np.sum(mu_hat ** 2)

    alpha_hat = num / den if den > 0 else alpha_init
    alpha_hat = max(alpha_hat, min_alpha)

    return alpha_hat, mu_hat, glm_nb_init

from sklearn.model_selection import TimeSeriesSplit

def rmse(y_true, y_pred):
    return np.sqrt(np.mean((y_true - y_pred)**2))


def tune_lambda_poisson_ridge(X, y, lambda_grid, n_splits=5):
    """
    Time-series cross-validation for ridge-penalized Poisson GLM.
    X, y must already be ordered in time.
    """
    tscv = TimeSeriesSplit(n_splits=n_splits)
    results = []

    for lam in lambda_grid:
        fold_rmses = []

        for train_idx, val_idx in tscv.split(X):
            X_tr, X_val = X[train_idx], X[val_idx]
            y_tr, y_val = y[train_idx], y[val_idx]

            glm_pois = sm.GLM(y_tr, X_tr, family=sm.families.Poisson())

            pen_weight = np.ones(X_tr.shape[1])
            pen_weight[0] = 0.0  # don't penalize intercept

            model = glm_pois.fit_regularized(
                method="elastic_net",
                alpha=lam,
                L1_wt=0.0,          # 0 => pure L2
                pen_weight=pen_weight,
                maxiter=200
            )

            y_pred = model.predict(X_val)
            fold_rmses.append(rmse(y_val, y_pred))

        mean_rmse = np.mean(fold_rmses)
        results.append((lam, mean_rmse))
        print(f"Poisson ridge: lambda={lam:.4g}, CV RMSE={mean_rmse:.3f}")

    best_lambda, best_rmse = min(results, key=lambda x: x[1])
    print("\nBest λ (Poisson ridge):", best_lambda, " with CV RMSE =", best_rmse)
    return best_lambda, best_rmse, results

def estimate_alpha_moment_nb2_from_poisson(y, X, min_alpha=1e-6):
    """
    Estimate NB2 dispersion alpha via method of moments, using a *Poisson* GLM
    to get a stable provisional mû.

    Var(Y|X) for NB2:  Var = mu + alpha * mu^2
    Method-of-moments alpha:
        alpha = sum((y - mu)^2 - mu) / sum(mu^2)
    """
    # stable provisional mean via Poisson
    glm_pois = sm.GLM(y, X, family=sm.families.Poisson()).fit()
    mu_hat = glm_pois.fittedvalues

    # avoid zero mu for safety
    mu_hat = np.clip(mu_hat, 1e-8, np.inf)

    resid_sq = (y - mu_hat)**2
    num = np.sum(resid_sq - mu_hat)
    den = np.sum(mu_hat**2)

    if den <= 0:
        alpha_hat = min_alpha
    else:
        alpha_hat = num / den
        if alpha_hat < min_alpha:
            alpha_hat = min_alpha

    return alpha_hat, mu_hat, glm_pois

def fit_hurdle_log_ridge_fold(X_tr, y_tr, lam_bin=1.0, lam_cnt=1.0):
    """
    Fit hurdle model on a *training fold*:
      - Binary part: ridge Binomial GLM, y_bin = 1(y>0)
      - Positive part: ridge Gaussian GLM on log(y), for y>0

    Returns the two fitted models.
    """
    # Binary outcome
    y_bin_tr = (y_tr > 0).astype(int)

    glm_bin = sm.GLM(y_bin_tr, X_tr, family=sm.families.Binomial())
    pen_weight_bin = np.ones(X_tr.shape[1])
    pen_weight_bin[0] = 0.0

    logit_ridge = glm_bin.fit_regularized(
        method="elastic_net",
        alpha=lam_bin,
        L1_wt=0.0,
        pen_weight=pen_weight_bin,
        maxiter=200
    )

    # Positive part (log y)
    mask_pos_tr = y_tr > 0
    if mask_pos_tr.sum() == 0:
        # Degenerate case: no positives in this fold (unlikely here)
        return logit_ridge, None

    y_pos_log_tr = np.log(y_tr[mask_pos_tr])
    X_pos_tr = X_tr[mask_pos_tr, :]

    glm_gauss = sm.GLM(y_pos_log_tr, X_pos_tr, family=sm.families.Gaussian())
    pen_weight_cnt = np.ones(X_pos_tr.shape[1])
    pen_weight_cnt[0] = 0.0

    gauss_ridge = glm_gauss.fit_regularized(
        method="elastic_net",
        alpha=lam_cnt,
        L1_wt=0.0,
        pen_weight=pen_weight_cnt,
        maxiter=200
    )

    return logit_ridge, gauss_ridge


def tune_hurdle_log_ridge(X, y, lam_bin_grid, lam_cnt_grid, n_splits=5):
    """
    Time-series CV for the hurdle model with:
      - Binomial(logit) part with λ_bin
      - Gaussian(log-positive) part with λ_cnt

    Returns the (λ_bin, λ_cnt) pair with smallest CV RMSE.
    """
    tscv = TimeSeriesSplit(n_splits=n_splits)
    results = []

    for lam_bin in lam_bin_grid:
        for lam_cnt in lam_cnt_grid:

            fold_rmses = []

            for train_idx, val_idx in tscv.split(X):
                X_tr, X_val = X[train_idx], X[val_idx]
                y_tr, y_val = y[train_idx], y[val_idx]

                logit_ridge, gauss_ridge = fit_hurdle_log_ridge_fold(
                    X_tr, y_tr,
                    lam_bin=lam_bin,
                    lam_cnt=lam_cnt
                )

                # P(Y>0 | X_val)
                p_pos_val = logit_ridge.predict(X_val)

                # If no positive model was fit, fallback: treat mu_pos=1
                if gauss_ridge is None:
                    mu_pos_val = np.ones_like(p_pos_val)
                else:
                    log_mu_pos_val = gauss_ridge.predict(X_val)
                    mu_pos_val = np.exp(log_mu_pos_val)

                mu_hurdle_val = p_pos_val * mu_pos_val

                fold_rmses.append(rmse(y_val, mu_hurdle_val))

            mean_rmse = np.mean(fold_rmses)
            results.append(((lam_bin, lam_cnt), mean_rmse))
            print(f"[Hurdle log] λ_bin={lam_bin:.4g}, λ_cnt={lam_cnt:.4g}, CV RMSE={mean_rmse:.3f}")

    (best_lam_bin, best_lam_cnt), best_rmse = min(results, key=lambda x: x[1])
    print(
        "\nBest (λ_bin, λ_cnt) (Hurdle log):",
        (best_lam_bin, best_lam_cnt),
        "  CV RMSE =", best_rmse
    )
    return best_lam_bin, best_lam_cnt, best_rmse, results
